def sha256(value):

def fsync_dir(path):
    fd=os.open(path,os.O_RDONLY|getattr(os,"O_DIRECTORY",0)|getattr(os,"O_NOFOLLOW",0))

        st=os.fstat(fd)

def exact_private_dir(path):

def ensure_private_child(parent,name):

def exact_file(path,maximum=MAX_RECORD_BYTES,allow_links=(1,)):
    before=os.lstat(path)
    if not stat.S_ISREG(before.st_mode) or stat.S_ISLNK(before.st_mode) or before.st_uid != os.getuid() or stat.S_IMODE(before.st_mode) != 0o600 or before.st_nlink not in allow_links or before.st_size < 1 or before.st_size > maximum:
        fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_FILE_INVALID")
    fd=os.open(path,os.O_RDONLY|getattr(os,"O_NOFOLLOW",0))
    try:
        opened=os.fstat(fd)
        if (opened.st_dev,opened.st_ino,opened.st_mode,opened.st_nlink,opened.st_size) != (before.st_dev,before.st_ino,before.st_mode,before.st_nlink,before.st_size):
            fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_FILE_INVALID")
        value=b""
        while len(value) < opened.st_size:
            part=os.read(fd,opened.st_size-len(value))

            value+=part
        after=os.fstat(fd); path_after=os.lstat(path)

        if fields(after) != fields(opened) or fields(path_after) != fields(opened):
            fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_FILE_INVALID")
        return value,opened
    finally:
        os.close(fd)

def parse_record_bytes(value,expected_name):
    try: raw=exact_keys(json.loads(value.decode("utf-8")),RECORD_KEYS)
    except Exception: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RECORD_INVALID")
    if raw["schemaVersion"] != 1 or raw["profile"] != PROFILE or RUN_ID.fullmatch(raw["runId"] or "") is None or not isinstance(raw["ordinal"],int) or isinstance(raw["ordinal"],bool) or not 1 <= raw["ordinal"] <= 99999999 or raw["phase"] not in PHASES:
        fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RECORD_INVALID")
    if expected_name != f'{raw["runId"]}-{raw["ordinal"]:08d}-{raw["phase"]}.json': fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RECORD_INVALID")
    if (raw["ordinal"] == 1) != (raw["previousRecordSha256"] is None) or (raw["previousRecordSha256"] is not None and SHA256.fullmatch(raw["previousRecordSha256"]) is None): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RECORD_INVALID")
    if not isinstance(raw["pid"],int) or isinstance(raw["pid"],bool) or raw["pid"] < 1 or not isinstance(raw["processStartIdentity"],str) or not 8 <= len(raw["processStartIdentity"]) <= 160 or any(ord(c) < 32 or ord(c) > 126 for c in raw["processStartIdentity"]): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RECORD_INVALID")
    if any(COMMIT.fullmatch(raw[key] or "") is None for key in ("protectedMain","moduleTree","bootstrapBlob")) or SHA256.fullmatch(raw["bootstrapSha256"] or "") is None or COMMAND.fullmatch(raw["command"] or "") is None or SHA256.fullmatch(raw["commandArgumentsSha256"] or "") is None: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RECORD_INVALID")
    root_absent=raw["runDev"] is None and raw["runIno"] is None
    root_present=isinstance(raw["runDev"],str) and DECIMAL.fullmatch(raw["runDev"]) and isinstance(raw["runIno"],str) and DECIMAL.fullmatch(raw["runIno"])
    if not (root_absent or root_present) or (raw["phase"] == "allocated" and not root_absent) or (raw["phase"] in ("launch-installing","launch-installed") and not root_present): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RECORD_INVALID")
    if raw["launchRecordSha256"] is not None and SHA256.fullmatch(raw["launchRecordSha256"]) is None: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RECORD_INVALID")
    if raw["phase"] == "launch-installed" and raw["launchRecordSha256"] is None: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RECORD_INVALID")
    containment_absent=raw["containedChildPid"] is None and raw["containedChildProcessStartIdentity"] is None and raw["containedChildPgid"] is None and raw["containmentConfirmationSha256"] is None
    containment_present=isinstance(raw["containedChildPid"],int) and not isinstance(raw["containedChildPid"],bool) and raw["containedChildPid"] > 1 and isinstance(raw["containedChildProcessStartIdentity"],str) and 1 <= len(raw["containedChildProcessStartIdentity"]) <= 160 and raw["containedChildPgid"] == raw["containedChildPid"] and isinstance(raw["containmentConfirmationSha256"],str) and SHA256.fullmatch(raw["containmentConfirmationSha256"]) is not None
    if not (containment_absent or containment_present) or (raw["phase"] in ("containment-prepared","operator-contained") and not containment_present): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RECORD_INVALID")
    cleanup=raw["phase"] in ("cleanup-prepared","tree-removing","run-removed","complete")
    if cleanup != (isinstance(raw["cleanupConfirmationSha256"],str) and SHA256.fullmatch(raw["cleanupConfirmationSha256"]) is not None): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RECORD_INVALID")
    if cleanup != (isinstance(raw["cleanupTreeInventorySha256"],str) and SHA256.fullmatch(raw["cleanupTreeInventorySha256"]) is not None): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RECORD_INVALID")
    if cleanup != (raw["cleanupReason"] in ("confirmed-dead-owner","completed-current-owner")): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RECORD_INVALID")
    return raw

def publish_record(authority,record):
    chain=read_authority(authority,record["runId"])
    latest=chain["records"][-1] if chain["records"] else None
    if record["ordinal"] != (1 if latest is None else latest["record"]["ordinal"]+1) or record["previousRecordSha256"] != (None if latest is None else latest["digest"]): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_PREDECESSOR_CAS_FAILED")
    value=canonical(record); digest=sha256(value)
    name=f'{record["runId"]}-{record["ordinal"]:08d}-{record["phase"]}.json'

def parse_terminal_bytes(value,run_id):

    return raw

def read_terminal(authority,run_id):
    final_path=os.path.join(authority,f"{run_id}-terminal.json")
    candidates=[]
    for name in os.listdir(authority):
        match=TERMINAL_TEMP.fullmatch(name)
        if match and match.group(1) == run_id: candidates.append((os.path.join(authority,name),match.group(2)))
    if len(candidates) > 1: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TERMINAL_INVALID")
    final=None; temporary=None
    if os.path.lexists(final_path):
        value,st=exact_file(final_path,allow_links=(1,2)); final={"path":final_path,"value":value,"stat":st,"digest":sha256(value),"terminal":parse_terminal_bytes(value,run_id)}
    if candidates:
        try:
            value,st=exact_file(candidates[0][0],allow_links=(1,2)); temporary={"path":candidates[0][0],"value":value,"stat":st,"digest":sha256(value),"terminal":parse_terminal_bytes(value,run_id),"partial":False}
            if temporary["digest"] != candidates[0][1]: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TERMINAL_INVALID")
        except Exception:
            st=os.lstat(candidates[0][0])
            if not stat.S_ISREG(st.st_mode) or stat.S_ISLNK(st.st_mode) or st.st_uid != os.getuid() or stat.S_IMODE(st.st_mode) != 0o600 or st.st_nlink not in (1,2) or st.st_size > MAX_RECORD_BYTES: raise
            temporary={"path":candidates[0][0],"stat":st,"digest":candidates[0][1],"terminal":None,"partial":True}
    if final and temporary:
        if temporary.get("partial"):
            return {**final,"publication":"installed-with-partial","temporaryPath":temporary["path"],"partialTemporary":True}
        if final["digest"] != temporary["digest"] or (final["stat"].st_dev,final["stat"].st_ino,final["stat"].st_nlink) != (temporary["stat"].st_dev,temporary["stat"].st_ino,2): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TERMINAL_INVALID")
        return {**final,"publication":"linked","temporaryPath":temporary["path"]}
    if final:
        if final["stat"].st_nlink != 1: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TERMINAL_INVALID")
        return {**final,"publication":"installed","temporaryPath":None}
    if temporary:
        if temporary.get("partial"):
            return {**temporary,"publication":"partial","finalPath":final_path,"temporaryPath":temporary["path"],"partialTemporary":True}
        if temporary["stat"].st_nlink != 1: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TERMINAL_INVALID")
        return {**temporary,"publication":"prelink","finalPath":final_path,"temporaryPath":temporary["path"]}
    return None

def repair_terminal(authority,run_id):
    terminal=read_terminal(authority,run_id)
    if terminal is None: return None
    if terminal["publication"] == "installed-with-partial":
        os.unlink(terminal["temporaryPath"]); fsync_dir(authority)
    elif terminal["publication"] == "partial": return terminal
    else:
        if terminal["publication"] == "prelink": os.link(terminal["temporaryPath"],terminal["finalPath"],follow_symlinks=False); fsync_dir(authority)
        if terminal["publication"] != "installed": os.unlink(terminal["temporaryPath"]); fsync_dir(authority)
    terminal=read_terminal(authority,run_id)
    if terminal is None or terminal["publication"] not in ("installed","partial"): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TERMINAL_INVALID")
    return terminal

def parse_launch(value):
    try: raw=exact_keys(json.loads(value.decode("utf-8")),LAUNCH_KEYS)
    except Exception: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
    if raw["schemaVersion"] != 1 or raw["profile"] != LAUNCH_PROFILE or raw["phase"] not in LAUNCH_PHASES or not isinstance(raw["pid"],int) or isinstance(raw["pid"],bool) or raw["pid"] < 1 or not isinstance(raw["processStartIdentity"],str) or not 8 <= len(raw["processStartIdentity"]) <= 160 or any(ord(c) < 32 or ord(c) > 126 for c in raw["processStartIdentity"]): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
    if DECIMAL.fullmatch(raw["runDev"] or "") is None or DECIMAL.fullmatch(raw["runIno"] or "") is None or any(COMMIT.fullmatch(raw[key] or "") is None for key in ("protectedMain","moduleTree","bootstrapBlob")) or SHA256.fullmatch(raw["bootstrapSha256"] or "") is None or COMMAND.fullmatch(raw["command"] or "") is None or SHA256.fullmatch(raw["commandArgumentsSha256"] or "") is None: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
    absent=raw["childPid"] is None and raw["childProcessStartIdentity"] is None and raw["childPgid"] is None
    present=isinstance(raw["childPid"],int) and not isinstance(raw["childPid"],bool) and raw["childPid"] > 1 and isinstance(raw["childProcessStartIdentity"],str) and 1 <= len(raw["childProcessStartIdentity"]) <= 160 and raw["childPgid"] == raw["childPid"]
    if not (absent or present): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
    if (raw["phase"] in ("launch-prepared","bootstrap-validating","operator-starting")) != absent: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
    if raw["terminal"] is None:
        if raw["phase"] in ("operator-terminal","complete"): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
    else:
        terminal=exact_keys(raw["terminal"],("code","signal","interruptedBy","reason"))
        if raw["phase"] not in ("operator-terminal","complete") or not (terminal["code"] is None or (isinstance(terminal["code"],int) and not isinstance(terminal["code"],bool))) or terminal["signal"] not in (None,"SIGINT","SIGTERM","SIGKILL") or terminal["interruptedBy"] not in (None,"SIGINT","SIGTERM") or terminal["reason"] not in ("operator-exit","bootstrap-interrupt","timeout") or (terminal["reason"] == "bootstrap-interrupt") != (terminal["interruptedBy"] is not None and terminal["reason"] != "timeout"): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
        if raw["phase"] == "complete" and terminal != {"code":0,"signal":None,"interruptedBy":None,"reason":"operator-exit"}: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
    return raw

def process_state(pid,start=None,start_digest=None):
    try: os.kill(pid,0)
    except ProcessLookupError: return "dead"
    except PermissionError: return "live-uninspectable"
    result=subprocess.run(["/bin/ps","-o","lstart=","-p",str(pid)],stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.DEVNULL,env={"PATH":"/usr/bin:/bin"},check=False,text=True)
    actual=result.stdout.strip()
    if result.returncode != 0 or not actual: return "live-uninspectable"
    if start is not None and actual == start: return "live-exact"
    if start_digest is not None and sha256(actual.encode()) == start_digest: return "live-exact"
    return "live-reused"

def group_state(pgid):
    if pgid is None: return "absent"
    try: os.killpg(pgid,0); return "live"
    except ProcessLookupError: return "absent"
    except PermissionError: return "live-uninspectable"

def launch_publication(run_root):
    if not os.path.lexists(run_root): return {"state":"absent","record":None,"digest":None,"path":None}
    exact_private_dir(run_root)
    final_path=os.path.join(run_root,"launch-record.json")
    temps=[]
    for name in os.listdir(run_root):
        match=LAUNCH_TEMP.fullmatch(name)
        if match: temps.append((name,match.group(1),match.group(2)))
    if len(temps) > len(LAUNCH_PHASES): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
    final=None; parsed_temps=[]; partial_temps=[]
    if os.path.lexists(final_path):
        try:
            value,st=exact_file(final_path,allow_links=(1,2)); final={"value":value,"stat":st,"record":parse_launch(value),"digest":sha256(value),"path":final_path}
        except Exception:
            st=os.lstat(final_path)
            return {"state":"partial-final","record":None,"digest":None,"path":final_path,"partial":{"dev":str(st.st_dev),"ino":str(st.st_ino),"nlink":st.st_nlink,"size":st.st_size},"temporaries":[]}
    for name,phase,expected_digest in temps:
        path=os.path.join(run_root,name)
        try:
            value,st=exact_file(path,allow_links=(1,2)); temp={"value":value,"stat":st,"record":parse_launch(value),"digest":sha256(value),"path":path,"phase":phase}
            if temp["digest"] != expected_digest or temp["record"]["phase"] != phase: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
            parsed_temps.append(temp)
        except Exception:
            st=os.lstat(path)
            if not stat.S_ISREG(st.st_mode) or stat.S_ISLNK(st.st_mode) or st.st_uid != os.getuid() or stat.S_IMODE(st.st_mode) != 0o600 or st.st_nlink not in (1,2) or st.st_size > MAX_RECORD_BYTES: raise
            partial_temps.append({"path":path,"phase":phase,"expectedDigest":expected_digest,"dev":str(st.st_dev),"ino":str(st.st_ino),"nlink":st.st_nlink,"size":st.st_size})
    if final:
        linked=[temp for temp in parsed_temps if (temp["stat"].st_dev,temp["stat"].st_ino) == (final["stat"].st_dev,final["stat"].st_ino)]
        if len(linked) > 1: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
        if linked:
            temp=linked[0]
            if final["digest"] != temp["digest"] or final["stat"].st_nlink != 2 or temp["stat"].st_nlink != 2: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
            updates=[item for item in parsed_temps if item is not temp]
            return {"state":"linked","record":final["record"],"digest":final["digest"],"path":final_path,"temporaryPath":temp["path"],"updateTemporaries":[{"path":item["path"],"phase":item["phase"],"digest":item["digest"]} for item in updates],"partialTemporaries":partial_temps}
        if final["stat"].st_nlink != 1: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
        return {"state":"installed" if not parsed_temps and not partial_temps else "installed-with-update-temporaries","record":final["record"],"digest":final["digest"],"path":final_path,"updateTemporaries":[{"path":item["path"],"phase":item["phase"],"digest":item["digest"]} for item in parsed_temps],"partialTemporaries":partial_temps}
    if len(parsed_temps) == 1 and not partial_temps:
        temp=parsed_temps[0]
        if temp["phase"] != "launch-prepared" or temp["stat"].st_nlink != 1: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
        return {"state":"prelink","record":temp["record"],"digest":temp["digest"],"path":temp["path"],"finalPath":final_path,"updateTemporaries":[],"partialTemporaries":[]}
    if not parsed_temps and len(partial_temps) == 1:
        return {"state":"partial","record":None,"digest":partial_temps[0]["expectedDigest"],"path":partial_temps[0]["path"],"partial":partial_temps[0],"updateTemporaries":[],"partialTemporaries":partial_temps}
    if parsed_temps or partial_temps: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_LAUNCH_RECORD_INVALID")
    return {"state":"missing","record":None,"digest":None,"path":None,"updateTemporaries":[],"partialTemporaries":[]}

def tree_inventory(run_root):

    def walk(directory_fd,relative,depth):

        opened=os.fstat(root_fd)
        if (str(opened.st_dev),str(opened.st_ino)) != (expected_dev,expected_ino) or opened.st_uid != os.getuid() or stat.S_IMODE(opened.st_mode) != 0o700: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RUN_IDENTITY_MISMATCH")
        def remove_children(directory_fd,depth,top):
            if depth > MAX_TREE_DEPTH: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_INVENTORY_BOUNDED")
            names=sorted(os.listdir(directory_fd),key=lambda name:(top and name == "launch-record.json",name))
            for name in names:
                if name in (".","..") or "/" in name or "\x00" in name: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TREE_INVALID")
                item=os.stat(name,dir_fd=directory_fd,follow_symlinks=False)
                if item.st_uid != os.getuid(): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TREE_INVALID")
                counts["entryCount"]+=1
                if counts["entryCount"] > MAX_TREE_ENTRIES: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_INVENTORY_BOUNDED")
                if stat.S_ISDIR(item.st_mode) and not stat.S_ISLNK(item.st_mode):
                    if stat.S_IMODE(item.st_mode) != 0o700: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TREE_INVALID")
                    child_fd=os.open(name,os.O_RDONLY|getattr(os,"O_DIRECTORY",0)|getattr(os,"O_NOFOLLOW",0),dir_fd=directory_fd)
                    try:
                        child=os.fstat(child_fd)
                        if (child.st_dev,child.st_ino,child.st_mode,child.st_uid) != (item.st_dev,item.st_ino,item.st_mode,item.st_uid): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TREE_CHANGED")
                        remove_children(child_fd,depth+1,False)
                    finally: os.close(child_fd)
                    after=os.stat(name,dir_fd=directory_fd,follow_symlinks=False)
                    if (after.st_dev,after.st_ino) != (item.st_dev,item.st_ino): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TREE_CHANGED")
                    os.rmdir(name,dir_fd=directory_fd); os.fsync(directory_fd)
                elif stat.S_ISREG(item.st_mode) and not stat.S_ISLNK(item.st_mode):
                    if stat.S_IMODE(item.st_mode) not in (0o400,0o500,0o600,0o644,0o700) or item.st_nlink != 1: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TREE_INVALID")
                    counts["byteCount"]+=item.st_size
                    if counts["byteCount"] > MAX_TREE_BYTES: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_INVENTORY_BOUNDED")
                    os.unlink(name,dir_fd=directory_fd); os.fsync(directory_fd)
                elif stat.S_ISLNK(item.st_mode):
                    if item.st_nlink != 1: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TREE_INVALID")
                    os.unlink(name,dir_fd=directory_fd); os.fsync(directory_fd)
                else: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TREE_INVALID")
        remove_children(root_fd,0,True)
        os.close(root_fd); root_fd=None
        current=os.stat(run_id,dir_fd=parent_fd,follow_symlinks=False)
        if (str(current.st_dev),str(current.st_ino)) != (expected_dev,expected_ino): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_RUN_IDENTITY_MISMATCH")
        os.rmdir(run_id,dir_fd=parent_fd); os.fsync(parent_fd)
        return counts
    finally:
        if root_fd is not None: os.close(root_fd)
        os.close(parent_fd)

def compact_complete_authority(authority,runs,run_id):
    chain=repair_authority(authority,run_id)
    if not chain["records"] or chain["records"][-1]["record"]["phase"] != "complete" or os.path.lexists(os.path.join(runs,run_id)): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_COMPACTION_INVALID")
    complete=chain["records"][-1]
    terminal=repair_terminal(authority,run_id)
    if terminal is not None and terminal["publication"] == "partial":
        os.unlink(terminal["temporaryPath"]); fsync_dir(authority); terminal=None
    if terminal is None: publish_terminal(authority,complete)
    repair_terminal(authority,run_id)
    chain=read_authority(authority,run_id)
    for entry in chain["records"]:
        if entry.get("terminal"): continue
        if os.path.lexists(entry["finalPath"]): os.unlink(entry["finalPath"]); fsync_dir(authority)
    terminal=repair_terminal(authority,run_id)
    if terminal is None: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_COMPACTION_INVALID")
    return terminal

def prune_terminal_tombstones(located,protected_run_id=None):
    _,runs,authority=located
    names=os.listdir(authority)
    finals=[]
    for name in names:
        match=TERMINAL_FINAL.fullmatch(name)
        if match and match.group(1) != protected_run_id:
            run_id=match.group(1)
            if os.path.lexists(os.path.join(runs,run_id)): continue
            st=os.lstat(os.path.join(authority,name))
            finals.append((st.st_mtime_ns,run_id))
    excess=max(0,len(finals)-MAX_TERMINAL_TOMBSTONES)
    removed=0
    for _,run_id in sorted(finals):
        if removed >= excess: break
        lock=try_acquire_lifecycle_lock(located,run_id)
        if lock is None: continue
        lock_path,lock_fd,lock_dev,lock_ino=lock
        try:
            terminal=repair_terminal(authority,run_id)
            chain=read_authority(authority,run_id)
            if os.path.lexists(os.path.join(runs,run_id)) or terminal is None or terminal["publication"] != "installed" or len(chain["records"]) != 1 or not chain["records"][0].get("terminal"): continue
            os.unlink(terminal["path"]); fsync_dir(authority); removed+=1
        finally: release_lifecycle_lock(authority,lock_path,lock_fd,lock_dev,lock_ino)

def cleanup_command(args):
    if len(args) != 3 or RUN_ID.fullmatch(args[1]) is None or SHA256.fullmatch(args[2]) is None: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_ARGUMENTS_INVALID")
    home,run_id,confirmation=args
    located=roots(home,False)
    if located is None: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_CONFIRMATION_REQUIRED")
    admin,runs,authority=located
    if not os.path.lexists(authority): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_CONFIRMATION_REQUIRED")
    inspected=inspect_one(admin,runs,authority,run_id,True)
    if inspected["confirmationDigest"] != confirmation: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_CONFIRMATION_REQUIRED")
    run_root=os.path.join(runs,run_id)
    if tree_inventory(run_root)["digest"] != inspected["treeInventory"]["digest"]: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TREE_CHANGED")
    chain=read_authority(authority,run_id); latest=chain["records"][-1] if chain["records"] else None
    if latest is not None and latest["record"]["phase"] == "complete":
        compact_complete_authority(authority,runs,run_id)
        prune_terminal_tombstones(located,run_id)
        print(json.dumps({"profile":PROFILE,"runId":run_id,"outcome":"already-complete","confirmationDigest":confirmation},sort_keys=True,separators=(",",":"))); return
    # Invalid prelink bytes have no authority until published. Once the exact
    # recorded owner is dead, they may be removed only under this confirmation.
    for partial in chain["partials"]:
        if process_state(partial["pid"],start_digest=partial["processStartIdentitySha256"]) in ("live-exact","live-uninspectable"): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_CLEANUP_BLOCKED")
        os.unlink(partial["path"]); fsync_dir(authority)
    if latest is None and not os.path.lexists(run_root):
        print(json.dumps({"profile":PROFILE,"runId":run_id,"outcome":"cleared-partial-authority","confirmationDigest":confirmation},sort_keys=True,separators=(",",":"))); return
    chain=repair_authority(authority,run_id); latest=chain["records"][-1] if chain["records"] else None
    launch=repair_launch_publication(run_root) if os.path.lexists(run_root) else launch_publication(run_root)
    prepared_inventory=tree_inventory(run_root)
    if inspected["containmentEligible"] or (latest is not None and latest["record"]["phase"] == "containment-prepared"):
        if latest is None: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_CONTAINMENT_AMBIGUOUS")
        if latest["record"]["phase"] == "launch-installed":
            if launch["record"] is None or launch["record"]["childPid"] is None or process_state(launch["record"]["childPid"],launch["record"]["childProcessStartIdentity"]) not in ("live-exact","dead") or group_state(launch["record"]["childPgid"]) != "live": fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_CONTAINMENT_AMBIGUOUS")
            latest=prepare_containment(authority,latest,launch,confirmation)
        if latest["record"]["phase"] != "containment-prepared": fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_CONTAINMENT_AMBIGUOUS")
        latest=finish_containment(authority,run_root,latest)
        print(json.dumps({"profile":PROFILE,"runId":run_id,"outcome":"operator-contained","containmentConfirmationDigest":latest["record"]["containmentConfirmationSha256"],"deletion":"blocked-pending-fresh-inspect-and-cutover-recovery"},sort_keys=True,separators=(",",":"))); return
    if not inspected["deletionEligible"]: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_CLEANUP_BLOCKED")
    source=latest["record"] if latest is not None else launch["record"]
    if source is None: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_CLEANUP_BLOCKED")
    deletion_in_progress=latest is not None and latest["record"]["phase"] in ("cleanup-prepared","tree-removing","run-removed")
    if not deletion_in_progress:
        if process_state(source["pid"],source["processStartIdentity"]) in ("live-exact","live-uninspectable"): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_CLEANUP_BLOCKED")
        if launch["record"] is not None and launch["record"]["childPid"] is not None:
            child_state=process_state(launch["record"]["childPid"],launch["record"]["childProcessStartIdentity"])
            if child_state in ("live-exact","live-uninspectable") or group_state(launch["record"]["childPgid"]) != "absent": fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_CLEANUP_BLOCKED")
        if global_blockers(admin): fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_CLEANUP_BLOCKED")
    newly_prepared=False
    if latest is None or latest["record"]["phase"] not in ("cleanup-prepared","tree-removing","run-removed"):
        run_st=exact_private_dir(run_root) if os.path.lexists(run_root) else None
        current_launch_digest=launch["digest"] if launch["record"] is not None else None
        if latest is None:
            record=base_record(run_id,1,"cleanup-prepared",None,source["pid"],source["processStartIdentity"],source["protectedMain"],source["moduleTree"],source["bootstrapBlob"],source["bootstrapSha256"],source["command"],source["commandArgumentsSha256"],run_st,current_launch_digest,confirmation,"confirmed-dead-owner",prepared_inventory["digest"])
            digest=publish_record(authority,record); latest={"record":record,"digest":digest,"publication":"installed","finalPath":os.path.join(authority,f'{run_id}-00000001-cleanup-prepared.json'),"temporaryPath":None}
        else:
            latest=append_phase(authority,latest,"cleanup-prepared",confirmation,"confirmed-dead-owner",run_st,current_launch_digest,prepared_inventory["digest"])
        newly_prepared=True
    if latest["record"]["phase"] == "cleanup-prepared":
        if tree_inventory(run_root)["digest"] != latest["record"]["cleanupTreeInventorySha256"]: fail("GREATER_REALM_PRODUCTION_LAUNCH_LIFECYCLE_TREE_CHANGED")
        latest=append_phase(authority,latest,"tree-removing",confirmation,"confirmed-dead-owner")
    if latest["record"]["phase"] == "tree-removing":
        if os.path.lexists(run_root): delete_run_tree(runs,run_id,latest["record"]["runDev"],latest["record"]["runIno"])
        latest=append_phase(authority,latest,"run-removed",confirmation,"confirmed-dead-owner")
    if latest["record"]["phase"] == "run-removed": latest=append_phase(authority,latest,"complete",confirmation,"confirmed-dead-owner")
    compact_complete_authority(authority,runs,run_id)
    prune_terminal_tombstones(located,run_id)
    print(json.dumps({"profile":PROFILE,"runId":run_id,"outcome":"cleaned","confirmationDigest":confirmation,"runState":"absent"},sort_keys=True,separators=(",",":")))
